void setup() {
  Serial.begin(9600);

  // Initialize the OLED with I2C address 0x3C (change to 0x3D if needed)
  if(!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("SSD1306 allocation failed"));
    for(;;);
  }

  // Clear the display buffer
  display.clearDisplay();

  // Draw the bitmap centered on the 128x64 screen
  // X = (128 - 32) / 2 = 48, Y = (64 - 32) / 2 = 16
  display.drawBitmap(48, 16, middle_finger_32x32, 32, 32, SSD1306_WHITE);
  
  // Push the buffer to the physical screen
  display.display();
}

  // Nothing to do here
}
